          canvas.style.width = canvasWidth + 'px';
          canvas.style.height = canvasHeight + 'px';
          ctx.setTransform(pixelRatio, 0, 0, pixelRatio, 0, 0);
        };

        resizeCanvas();
        window.addEventListener('resize', resizeCanvas, { passive: true });

        const addParticles = (x, y, amount, force = 1) => {
          for (let i = 0; i < amount; i++) {
            const angle = Math.random() * Math.PI * 2;
            const speed = (0.45 + Math.random() * 1.9) * force;
            particles.push({
              x,
              y,
              vx: Math.cos(angle) * speed,
              vy: Math.sin(angle) * speed - Math.random() * .8,
              life: 1,
              decay: .012 + Math.random() * .018,
              size: 1.2 + Math.random() * 2.5,
              color: particleColors[Math.floor(Math.random() * particleColors.length)]
            });
          }
          if (particles.length > 180) {
            particles.splice(0, particles.length - 180);
          }
          startParticleAnimation();
        };

        const animateParticles = () => {
          animationFrame = null;
          if (document.hidden || particles.length === 0) {
            ctx.clearRect(0, 0, canvasWidth, canvasHeight);
            return;
          }
          ctx.clearRect(0, 0, canvasWidth, canvasHeight);
          particles = particles.filter(particle => particle.life > 0);
          particles.forEach(particle => {
            particle.x += particle.vx;
            particle.y += particle.vy;
            particle.vx *= .985;
            particle.vy *= .985;
            particle.life -= particle.decay;
            ctx.beginPath();
            ctx.fillStyle = `rgba(${particle.color},${Math.max(0, particle.life) * .85})`;
            ctx.shadowBlur = 10;
            ctx.shadowColor = `rgba(${particle.color},.65)`;
            ctx.arc(particle.x, particle.y, particle.size * Math.max(0, particle.life), 0, Math.PI * 2);
            ctx.fill();
          });
          ctx.shadowBlur = 0;
          if (particles.length > 0) {
            animationFrame = requestAnimationFrame(animateParticles);
          }
        };

        function startParticleAnimation() {
          if (document.hidden || animationFrame !== null || particles.length === 0) return;
          animationFrame = requestAnimationFrame(animateParticles);
        }

        let lastMouseParticle = 0;
        document.addEventListener('pointermove', event => {
          if (event.pointerType !== 'mouse') return;
          const now = performance.now();
          if (now - lastMouseParticle < 30) return;
          lastMouseParticle = now;
          addParticles(event.clientX, event.clientY, 2, .65);
        }, { passive: true });

        let lastTouchParticle = 0;
        document.addEventListener('pointerdown', event => {
          if (event.pointerType === 'mouse') return;
          const now = performance.now();
          if (now - lastTouchParticle < 75) return;
          lastTouchParticle = now;
          addParticles(event.clientX, event.clientY, 12, 1.1);
        }, { passive: true });

        document.addEventListener('visibilitychange', () => {
          if (document.hidden) {
            if (animationFrame !== null) {
              cancelAnimationFrame(animationFrame);
              animationFrame = null;
            }
            ctx.clearRect(0, 0, canvasWidth, canvasHeight);
          }
        });
      }

      const contactModal = document.getElementById('contactModal');
      const openContact = document.getElementById('openContact');
      const closeContact = document.getElementById('closeContact');
      const copyEmail = document.getElementById('copyEmail');
      const copyFeedback = document.getElementById('copyFeedback');
      const contactEmailElement = document.getElementById('contactEmail');
      const contactEmail = contactEmailElement?.textContent.trim() || '';

      let previousFocus = null;

      const openContactModal = () => {
        if (!contactModal) return;
        previousFocus = document.activeElement;
        contactModal.classList.add('open');
        contactModal.setAttribute('aria-hidden', 'false');
        document.body.classList.add('modal-open');
        if (copyFeedback) copyFeedback.textContent = '';
        closeContact?.focus();
      };

      window.closeContactModal = function() {
        if (!contactModal) return;
        contactModal.classList.remove('open');
        contactModal.setAttribute('aria-hidden', 'true');
        document.body.classList.remove('modal-open');
        if (previousFocus && typeof previousFocus.focus === 'function') {
          previousFocus.focus();
        }
      };

      openContact?.addEventListener('click', openContactModal);
      closeContact?.addEventListener('click', closeContactModal);

      contactModal?.addEventListener('click', event => {
        if (event.target === contactModal) closeContactModal();
      });

      contactModal?.addEventListener('keydown', event => {
        if (event.key !== 'Tab') return;
        const focusable = contactModal.querySelectorAll(
          'button:not([disabled]), a[href], input:not([disabled]), [tabindex]:not([tabindex="-1"])'
        );
        if (!focusable.length) return;
        const first = focusable[0];
        const last = focusable[focusable.length - 1];
        if (event.shiftKey && document.activeElement === first) {
          event.preventDefault();
          last.focus();
        } else if (!event.shiftKey && document.activeElement === last) {
          event.preventDefault();
          first.focus();
        }
      });

      copyEmail?.addEventListener('click', async () => {
        if (!contactEmail) {
          if (copyFeedback) copyFeedback.textContent = 'Indirizzo email non disponibile.';
          return;
        }
        try {
          if (navigator.clipboard && window.isSecureContext) {
            try {
              await navigator.clipboard.writeText(contactEmail);
            } catch (clipboardError) {
              const temp = document.createElement('textarea');
              temp.value = contactEmail;
              temp.setAttribute('readonly', '');
              temp.style.position = 'fixed';
              temp.style.opacity = '0';
              document.body.appendChild(temp);
              temp.select();
              const copied = document.execCommand('copy');
              temp.remove();
              if (!copied) throw clipboardError;
            }
          } else {
            const temp = document.createElement('textarea');
            temp.value = contactEmail;
            temp.setAttribute('readonly', '');
            temp.style.position = 'fixed';
            temp.style.opacity = '0';
            document.body.appendChild(temp);
            temp.select();
            const copied = document.execCommand('copy');
            temp.remove();
            if (!copied) throw new Error('Copia non disponibile');
          }
          if (copyFeedback) copyFeedback.textContent = 'Email copiata correttamente!';
        } catch (error) {
          if (copyFeedback) copyFeedback.textContent = 'Copia non riuscita. Seleziona manualmente l’indirizzo.';
        }
      });

      const cookieKey = 'creatorweb_cookie_preferences';
      const cookieVersion = '1.0';
      const cookieBanner = document.getElementById('cookieBanner');
      const cookieMainPanel = document.getElementById('cookieMainPanel');
      const cookieSettingsPanel = document.getElementById('cookieSettingsPanel');
      const analyticsConsent = document.getElementById('analyticsConsent');

      const readCookiePreferences = () => {
        try {
          const saved = JSON.parse(localStorage.getItem(cookieKey) || 'null');
          if (!saved || saved.version !== cookieVersion || !saved.expiry || Date.now() > saved.expiry) return null;
          return saved;
        } catch (error) {
          return null;
        }
      };

      const openCookieBanner = (settings = false) => {
        if (!cookieBanner || !cookieMainPanel || !cookieSettingsPanel || !analyticsConsent) return;
        cookieBanner.hidden = false;
        cookieMainPanel.hidden = settings;
        cookieSettingsPanel.hidden = !settings;
        const saved = readCookiePreferences();
        analyticsConsent.checked = Boolean(saved?.analytics);
      };

      const saveCookiePreferences = analytics => {
        const preferences = {
          version: cookieVersion,
          necessary: true,
          analytics: Boolean(analytics),
          savedAt: new Date().toISOString(),
          expiry: Date.now() + 180 * 24 * 60 * 60 * 1000
        };
        try {
          localStorage.setItem(cookieKey, JSON.stringify(preferences));
        } catch (error) {}
        if (cookieBanner) cookieBanner.hidden = true;
        try {
          if (typeof window.CW_COOKIE_HOOKS?.onPreferenceChange === 'function') {
            window.CW_COOKIE_HOOKS.onPreferenceChange(preferences);
          }
        } catch (error) {
          console.error('Errore nel collegamento delle preferenze cookie:', error);
        }
      };

      document.getElementById('acceptCookies')?.addEventListener('click', () => saveCookiePreferences(true));
      document.getElementById('rejectCookies')?.addEventListener('click', () => saveCookiePreferences(false));
      document.getElementById('customizeCookies')?.addEventListener('click', () => openCookieBanner(true));
      document.getElementById('rejectSettingsCookies')?.addEventListener('click', () => saveCookiePreferences(false));
      document.getElementById('saveCookieSettings')?.addEventListener('click', () => saveCookiePreferences(analyticsConsent?.checked));
      document.getElementById('backCookieSettings')?.addEventListener('click', () => {
        if (cookieMainPanel && cookieSettingsPanel) {
          cookieMainPanel.hidden = false;
          cookieSettingsPanel.hidden = true;
        }
      });
      document.getElementById('openCookieSettings')?.addEventListener('click', () => openCookieBanner(true));

      window.CW_OPEN_COOKIE_SETTINGS = () => openCookieBanner(true);

      if (!readCookiePreferences()) {
        openCookieBanner(false);
      }
    })();
  </script>

  <script>
    (() => {
      'use strict';

      if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) return;

      const selector = [
        'h1', 'h2', 'h3', 'h4', 'h5', 'h6',
        '.panel', '.card', '.modal', 'article', '.faq-question'
      ].join(',');

      let items = [];
      let rafId = null;
      let lastScrollY = window.scrollY || window.pageYOffset || 0;
      let scrollVelocity = 0;

      function collect() {
        const nodes = [...document.querySelectorAll(selector)].filter(el => {
          if (el.closest('[hidden]') || el.closest('.intro-screen') || el.closest('.site-header')) {
            return false;
          }
          return el.getClientRects().length > 0;
        });

        items = nodes.map(el => {
          el.classList.add('scroll-scale-target');
          const currentScale = parseFloat(el.dataset.currentScale) || 1;
          return {
            el,
            targetScale: currentScale,
            currentScale
          };
        });
      }

      function smoothstep(edge0, edge1, x) {
        const t = Math.max(0, Math.min(1, (x - edge0) / (edge1 - edge0)));
        return t * t * (3 - 2 * t);
      }

      function computeTargetScale(centerY, viewportH) {
        if (!viewportH) return 1;
        const pos = centerY / viewportH;
        if (pos < 0.22) {
          const t = smoothstep(0, 0.22, pos);
          return 0.78 + 0.22 * t;
        }
        if (pos <= 0.86) {
          return 1;
        }
        const t = smoothstep(0.86, 1.18, pos);
        return 1 + 0.25 * t;
      }

      function measureTargets() {
        const vh = window.innerHeight || document.documentElement.clientHeight;
        for (let i = 0; i < items.length; i++) {
          const item = items[i];
          if (!item.el.isConnected) continue;
          const rect = item.el.getBoundingClientRect();
          const center = rect.top + rect.height * 0.5;
          item.targetScale = computeTargetScale(center, vh);
        }
      }

      function step() {
        measureTargets();
        let needsMoreFrames = false;
        const speedBoost = Math.min(0.45, Math.abs(scrollVelocity) * 0.008);
        const lerpFactor = Math.min(0.65, 0.16 + speedBoost);

        for (let i = 0; i < items.length; i++) {
          const item = items[i];
          const diff = item.targetScale - item.currentScale;
          if (Math.abs(diff) > 0.0012) {
            item.currentScale += diff * lerpFactor;
            needsMoreFrames = true;
          } else {
            item.currentScale = item.targetScale;
          }
          item.el.dataset.currentScale = item.currentScale.toFixed(4);
          item.el.style.transform = `scale3d(${item.currentScale.toFixed(4)}, ${item.currentScale.toFixed(4)}, 1)`;
        }

        scrollVelocity *= 0.82;

        if (needsMoreFrames) {
          rafId = requestAnimationFrame(step);
        } else {
          rafId = null;
        }
      }

      function onScroll() {
        const currentY = window.scrollY || window.pageYOffset || 0;
        scrollVelocity = currentY - lastScrollY;
        lastScrollY = currentY;
        if (rafId === null) {
          rafId = requestAnimationFrame(step);
        }
      }

      collect();
      if (rafId === null) {
        rafId = requestAnimationFrame(step);
      }

      window.addEventListener('scroll', onScroll, { passive: true });
      window.addEventListener('resize', () => {
        collect();
        if (rafId === null) rafId = requestAnimationFrame(step);
      }, { passive: true });

      document.getElementById('toggleMoreDemos')?.addEventListener('click', () => {
        setTimeout(() => {
          collect();
          if (rafId === null) rafId = requestAnimationFrame(step);
        }, 60);
      });
    })();
  </script>
